\[  1- \lvert x \rvert=\frac{1}{2} + \sum_{n=1}^{\infty} a_{n}\cos\bigl((2n-1)\pi x\bigr). \]

\[  \mathcal{B}(x;b)=(1- \lvert x \rvert)^{\beta(b)}=(\frac{1}{2} + \sum_{n=1}^{\infty} a_{n}\cos\bigl((2n-1)\pi x\bigr))^{\beta(b)}. \]
